\[
	A_{21}(P_2 - P_1) +	A_{23}(P_2 - P_3) = 0
\]
\[
	(A_{21} + A_{23})P_2 - A_{21} P_1 - A_{23}P_3 = 0
\]
\[
	P_1 = 100
\]
\[
	- A_{21} P_1 + (A_{21} + A_{23})P_2  - A_{23}P_3 = 0
\]
\[
	P_3 = 10
\]
\[
\begin{bmatrix}
    1 & 0 & 0 \\
    - A_{21} & (A_{21} + A_{23}) & -A_{23} \\
    0 & 0 & 1
\end{bmatrix}
\begin{bmatrix}
    P_1 \\
    P_2  \\
    P_3
\end{bmatrix}
=
\begin{bmatrix}
	100 \\
    0 \\
    10
\end{bmatrix}
\]
\[
	\mathbf{M} \vec{p} = \vec{b}
\]
\[
 M =
 \begin{bmatrix}
    1 & 0 & 0 \\
    - A_{21} & (A_{21} + A_{23}) & -A_{23} \\
    0 & 0 & 1
\end{bmatrix},
\quad
\vec{p} =
\begin{bmatrix}
    P_1 \\
    P_2  \\
    P_3
\end{bmatrix},
\quad
\vec{b}=
\begin{bmatrix}
	100 \\
    0 \\
    10
\end{bmatrix}
\]

	A_{21} = \frac{\pi R_{21}^4}{8 \mu L_{21}} = \frac{\pi (1)^4}{8 \mu (10)} = \num{0.039}
\]
\[
	A_{23} = \frac{\pi R_{23}^4}{8 \mu L_{23}} = \frac{\pi (0.7)^4}{8 \mu (8)} = \num{0.012}
\]
\[
	\vec{p} = \mathbf{M^{-1}} \vec{b}
\]
\[
\vec{p} =
{\begin{bmatrix}
    1 & 0 & 0 \\
    -\num{0.039} & \num{0.051} & -\num{0.012} \\

\end{bmatrix}}^{-1}
\begin{bmatrix}
	100 \\
    0 \\
    10
\end{bmatrix}

\vec{p} =
\begin{bmatrix}
	100 \\

    10
\end{bmatrix}
